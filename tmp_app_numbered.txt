15: // ImportMeta declarations are provided in resources/js/env.d.ts

17: const appName = import.meta.env.VITE_APP_NAME || 'Laravel';
18: 
19: // Configuracion centralizada de Axios para Laravel + Inertia
20: // SOLO usar XSRF-TOKEN cookie (Laravel lo genera automaticamente)
21: // NO fijar X-CSRF-TOKEN global - Inertia lo maneja por su cuenta
22: axios.defaults.headers.common['X-Requested-With'] = 'XMLHttpRequest';
23: axios.defaults.headers.common['Accept'] = 'application/json';
24: axios.defaults.xsrfCookieName = 'XSRF-TOKEN';
25: axios.defaults.xsrfHeaderName = 'X-XSRF-TOKEN';
26: axios.defaults.withCredentials = true; // Siempre enviar cookies (requerido para CSRF)
27: 
28: // Eliminar Content-Type de headers defaults para que axios detecte FormData automaticamente
29: // Inicializar objetos de headers si no existen, luego eliminar Content-Type
30: axios.defaults.headers.post = axios.defaults.headers.post || {};
31: axios.defaults.headers.put = axios.defaults.headers.put || {};
32: axios.defaults.headers.patch = axios.defaults.headers.patch || {};
33: delete axios.defaults.headers.post['Content-Type'];
34: delete axios.defaults.headers.put['Content-Type'];
35: delete axios.defaults.headers.patch['Content-Type'];
36: delete axios.defaults.headers.common['Content-Type'];
37: 
38: const axiosDebugEnabled = import.meta.env.DEV && import.meta.env.VITE_DEBUG_AXIOS === 'true';
39: 
40: // Interceptor para garantizar que FormData no tenga Content-Type forzado
41: axios.interceptors.request.use((config) => {
42:     const requestId = (config.headers?.['X-Request-Id'] as string | undefined) ?? undefined;
43: 
44:     if (axiosDebugEnabled) {
45:         console.debug('[AXIOS][REQ]', {
46:             method: config.method?.toUpperCase(),
47:             url: config.url,
48:             requestId: requestId ?? null,
49:             timestamp: new Date().toISOString(),
50:         });
51:     }
52: 
53:     if (config.data instanceof FormData) {
54:         // Eliminar Content-Type para que el browser lo detecte automaticamente con boundary
55:         delete config.headers['Content-Type'];
56:     }
57:     return config;
58: });
59: 
60: // NO fijar X-CSRF-TOKEN desde meta tag: Inertia y el cookie XSRF-TOKEN ya lo manejan.
61: // Configurar esto globalmente puede causar conflictos cuando el meta tag queda desactualizado.
62: 
63: axios.interceptors.response.use(
64:     (response) => {
65:         if (axiosDebugEnabled) {
66:             const requestId = (response.config.headers?.['X-Request-Id'] as string | undefined) ?? undefined;
67:             console.debug('[AXIOS][RES]', {
68:                 method: response.config.method?.toUpperCase(),
69:                 url: response.config.url,
70:                 status: response.status,
71:                 requestId: requestId ?? null,
72:                 timestamp: new Date().toISOString(),
73:             });
74:         }
75: 
76:         return response;
77:     },
78:     (error) => {
79:         if (axiosDebugEnabled) {
80:             const requestId = (error?.config?.headers?.['X-Request-Id'] as string | undefined) ?? undefined;
81:             console.debug('[AXIOS][ERR]', {
82:                 method: error?.config?.method?.toUpperCase(),
83:                 url: error?.config?.url,
84:                 status: error?.response?.status ?? null,
85:                 requestId: requestId ?? null,
86:                 timestamp: new Date().toISOString(),
87:             });
88:         }
89: 
90:         if (error?.response?.status === 401) {
91:             window.location.href = '/login';
92:         }
93:         return Promise.reject(error);
94:     },
95: );
96: 
97: createInertiaApp({
98:     title: (title) => `${title} - ${appName}`,
99:     resolve: (name) => resolvePageComponent(`./pages/${name}.vue`, import.meta.glob<DefineComponent>('./pages/**/*.vue')),
100:     setup({ el, App, props, plugin }) {
101:         const pinia = createPinia(); // Crear instancia de Pinia
102:         const smartLoading = useSmartLoading();

104: 
105:         createApp({
106:             render: () => h(Fragment, [h(App, props), h(GlobalLoadingOverlay)]),
107:         })
108:             .use(plugin)
109:             .use(pinia) // Usar Pinia
110:             .use(ZiggyVue)
111:             .mount(el);
112:     },
113:     progress: {
114:         color: '#4B5563',
115:     },
116: });

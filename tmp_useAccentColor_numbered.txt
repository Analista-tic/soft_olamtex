5: export const accentColors = ['zinc', 'red', 'rose', 'orange', 'green', 'blue', 'yellow', 'violet'] as const;
6: export type AccentColor = (typeof accentColors)[number];
7: 
8: interface AccentColorAuthUser {
9:     theme_color?: AccentColor | null;
10: }
11: 
12: interface AccentColorPageProps {
13:     auth?: {
14:         user?: AccentColorAuthUser | null;
15:     };
16: }
17: 
18: const DEFAULT_ACCENT_COLOR: AccentColor = 'blue';
19: const STORAGE_KEY = 'accentColor';
20: const COOKIE_KEY = 'accentColor';
21: 
22: const isAccentColor = (value: string | null): value is AccentColor => {
23:     return value !== null && accentColors.includes(value as AccentColor);
24: };
25: 
26: const setCookie = (name: string, value: string, days = 365) => {
27:     if (typeof document === 'undefined') {
28:         return;
29:     }
30: 
31:     const maxAge = days * 24 * 60 * 60;
32:     document.cookie = `${name}=${value};path=/;max-age=${maxAge};SameSite=Lax`;
33: };
34: 
35: const getStoredAccentColor = (): AccentColor => {
36:     if (typeof window === 'undefined') {
37:         return DEFAULT_ACCENT_COLOR;
38:     }
39: 
40:     const storedValue = localStorage.getItem(STORAGE_KEY);
41:     return isAccentColor(storedValue) ? storedValue : DEFAULT_ACCENT_COLOR;
42: };
43: 
44: const persistAccentColor = (value: AccentColor) => {
45:     if (typeof window === 'undefined') {
46:         return;
47:     }
48: 
49:     localStorage.setItem(STORAGE_KEY, value);
50:     setCookie(COOKIE_KEY, value);
51: };
52: 
53: export const applyAccentColor = (value: AccentColor) => {
54:     if (typeof document === 'undefined') {
55:         return;
56:     }
57: 
58:     document.documentElement.dataset.accent = value;
59: };
60: 
61: export const initAccentColor = () => {
62:     if (typeof window === 'undefined') {
63:         return;
64:     }
65: 
66:     const accentColor = getStoredAccentColor();
67:     persistAccentColor(accentColor);
68:     applyAccentColor(accentColor);
69: };

71: export function useAccentColor() {
72:     const page = usePage<AccentColorPageProps>();
73:     const accentColor = ref<AccentColor>(DEFAULT_ACCENT_COLOR);
74: 
75:     const initializeAccentColor = () => {
76:         if (typeof window === 'undefined') {
77:             return;
78:         }
79: 
80:         const userPref = page.props?.auth?.user?.theme_color;
81: 
82:         if (userPref && isAccentColor(userPref)) {
83:             accentColor.value = userPref;
84:             persistAccentColor(userPref);
85:             applyAccentColor(userPref);
86: 
87:             return;
88:         }
89: 
90:         const storedValue = getStoredAccentColor();
91: 
92:         accentColor.value = storedValue;
93:         applyAccentColor(storedValue);
94:     };
95: 
96:     onMounted(() => {
97:         initAccentColor();

99:     });
100: 
101:     const updateAccentColor = (value: AccentColor) => {
102:         accentColor.value = value;
103:         persistAccentColor(value);
104:         applyAccentColor(value);
105: 
106:         axios
107:             .patch(route('settings.appearance.update'), {
108:                 theme_color: value,
109:             })
110:             .catch(() => {
111:                 // La UI ya quedó aplicada localmente aunque falle la persistencia remota.
112:             });
113:     };
114: 
115:     return {

118:         updateAccentColor,

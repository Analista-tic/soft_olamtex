1: import { usePage } from '@inertiajs/vue3';
2: import axios from 'axios';
3: import { onMounted, ref } from 'vue';
4: 
5: type Appearance = 'light' | 'dark' | 'system';
6: const STORAGE_KEY = 'appearance';
7: 
8: interface AppearanceAuthUser {
9:     theme_appearance?: Appearance | null;
10: }
11: 
12: interface AppearancePageProps {
13:     auth?: {
14:         user?: AppearanceAuthUser | null;
15:     };
16: }
17: 
18: export function updateTheme(value: Appearance) {
19:     if (typeof window === 'undefined') {
20:         return;
21:     }
22: 
23:     if (value === 'system') {
24:         const mediaQueryList = window.matchMedia('(prefers-color-scheme: dark)');
25:         const systemTheme = mediaQueryList.matches ? 'dark' : 'light';
26: 
27:         document.documentElement.classList.toggle('dark', systemTheme === 'dark');
28:     } else {
29:         document.documentElement.classList.toggle('dark', value === 'dark');
30:     }
31: }
32: 
33: const setCookie = (name: string, value: string, days = 365) => {
34:     if (typeof document === 'undefined') {
35:         return;
36:     }
37: 
38:     const maxAge = days * 24 * 60 * 60;
39: 
40:     document.cookie = `${name}=${value};path=/;max-age=${maxAge};SameSite=Lax`;
41: };
42: 
43: const mediaQuery = () => {
44:     if (typeof window === 'undefined') {
45:         return null;
46:     }
47: 
48:     return window.matchMedia('(prefers-color-scheme: dark)');
49: };
50: 
51: const getStoredAppearance = () => {
52:     if (typeof window === 'undefined') {
53:         return null;
54:     }
55: 
56:     return localStorage.getItem(STORAGE_KEY) as Appearance | null;
57: };
58: 
59: const handleSystemThemeChange = () => {
60:     const currentAppearance = getStoredAppearance();
61: 
62:     updateTheme(currentAppearance || 'system');
63: };
64: 
65: export function initializeTheme() {
66:     if (typeof window === 'undefined') {
67:         return;
68:     }
69: 
70:     // Initialize theme from saved preference or default to system...
71:     const savedAppearance = getStoredAppearance();
72:     updateTheme(savedAppearance || 'system');
73: 
74:     // Set up system theme change listener...
75:     mediaQuery()?.addEventListener('change', handleSystemThemeChange);
76: }
77: 
78: export function useAppearance() {
79:     const page = usePage<AppearancePageProps>();
80:     const appearance = ref<Appearance>('system');
81: 
82:     function initializeAppearance() {
83:         if (typeof window === 'undefined') {
84:             return;
85:         }
86: 
87:         const userPref = page.props?.auth?.user?.theme_appearance;
88: 
89:         if (userPref) {
90:             appearance.value = userPref;
91:             localStorage.setItem(STORAGE_KEY, userPref);
92:             setCookie('appearance', userPref);
93:             updateTheme(userPref);
94: 
95:             return;
96:         }
97: 
98:         const savedAppearance = localStorage.getItem(STORAGE_KEY) as Appearance | null;
99:         const resolvedAppearance: Appearance = savedAppearance || 'system';
100: 
101:         appearance.value = resolvedAppearance;
102:         updateTheme(resolvedAppearance);
103:     }
104: 
105:     onMounted(() => {
106:         initializeTheme();
107:         initializeAppearance();
108:     });
109: 
110:     function updateAppearance(value: Appearance) {
111:         appearance.value = value;
112: 
113:         // Store in localStorage for client-side persistence...
114:         localStorage.setItem(STORAGE_KEY, value);
115: 
116:         // Store in cookie for SSR...
117:         setCookie('appearance', value);
118: 
119:         updateTheme(value);
120: 
121:         axios
122:             .patch(route('settings.appearance.update'), {
123:                 theme_appearance: value,
124:             })
125:             .catch(() => {
126:                 // La UI ya quedó aplicada localmente aunque falle la persistencia remota.
127:             });
128:     }
129: 
130:     return {
131:         appearance,
132:         updateAppearance,
133:     };
134: }
